
-- ==========================================
-- MEMBERKU CRM - DATABASE SETUP
-- Supabase PostgreSQL
-- ==========================================

-- 1. PROFILES (ADMIN)
CREATE TABLE public.profiles (
    id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    full_name TEXT NOT NULL,
    role TEXT NOT NULL DEFAULT 'admin'
        CHECK (role IN ('admin')),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. MEMBERS
CREATE TABLE public.members (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL,
    phone TEXT NOT NULL UNIQUE,
    email TEXT,
    points_balance INTEGER NOT NULL DEFAULT 0
        CHECK (points_balance >= 0),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. TRANSACTIONS
CREATE TABLE public.transactions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    member_id UUID NOT NULL REFERENCES public.members(id),
    admin_id UUID REFERENCES public.profiles(id),
    total_amount NUMERIC(15,2) NOT NULL CHECK (total_amount > 0),
    points_earned INTEGER NOT NULL DEFAULT 0,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. POINT TRANSACTIONS
CREATE TABLE public.point_transactions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    member_id UUID NOT NULL REFERENCES public.members(id),
    type TEXT NOT NULL CHECK (
        type IN ('earn', 'redeem', 'reversal', 'order_earn', 'order_reversal')
    ),
    points_delta INTEGER NOT NULL CHECK (points_delta <> 0),
    reference_type TEXT,
    reference_id UUID,
    description TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 5. PRODUCTS
CREATE TABLE public.products (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL,
    description TEXT,
    price NUMERIC(15,2) NOT NULL CHECK (price >= 0),
    image_url TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 6. ORDERS
CREATE TABLE public.orders (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    order_number TEXT NOT NULL UNIQUE,
    member_id UUID REFERENCES public.members(id) ON DELETE SET NULL,
    customer_name TEXT NOT NULL,
    customer_phone TEXT NOT NULL,
    total_amount NUMERIC(15,2) NOT NULL DEFAULT 0,
    status TEXT NOT NULL DEFAULT 'Baru'
        CHECK (status IN ('Baru', 'Diproses', 'Selesai', 'Dibatalkan')),
    points_awarded INTEGER NOT NULL DEFAULT 0,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 7. ORDER ITEMS
CREATE TABLE public.order_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    order_id UUID NOT NULL REFERENCES public.orders(id) ON DELETE CASCADE,
    product_id UUID NOT NULL REFERENCES public.products(id),
    quantity INTEGER NOT NULL CHECK (quantity > 0),
    unit_price NUMERIC(15,2) NOT NULL CHECK (unit_price >= 0),
    subtotal NUMERIC(15,2) GENERATED ALWAYS AS (quantity * unit_price) STORED
);

-- ==========================================
-- INDEXES
-- ==========================================

CREATE INDEX idx_members_name ON public.members(name);
CREATE INDEX idx_members_phone ON public.members(phone);
CREATE INDEX idx_transactions_member ON public.transactions(member_id);
CREATE INDEX idx_transactions_date ON public.transactions(created_at DESC);
CREATE INDEX idx_points_member ON public.point_transactions(member_id);
CREATE INDEX idx_orders_status ON public.orders(status);
CREATE INDEX idx_orders_date ON public.orders(created_at DESC);
CREATE INDEX idx_order_items_order ON public.order_items(order_id);

-- ==========================================
-- UPDATED AT TRIGGER
-- ==========================================

CREATE OR REPLACE FUNCTION public.set_updated_at()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$;

CREATE TRIGGER members_updated_at
BEFORE UPDATE ON public.members
FOR EACH ROW EXECUTE FUNCTION public.set_updated_at();

CREATE TRIGGER products_updated_at
BEFORE UPDATE ON public.products
FOR EACH ROW EXECUTE FUNCTION public.set_updated_at();

CREATE TRIGGER orders_updated_at
BEFORE UPDATE ON public.orders
FOR EACH ROW EXECUTE FUNCTION public.set_updated_at();

-- ==========================================
-- ADMIN CHECK FUNCTION
-- ==========================================

CREATE OR REPLACE FUNCTION public.is_admin()
RETURNS BOOLEAN
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $$
    SELECT EXISTS (
        SELECT 1
        FROM public.profiles
        WHERE id = (SELECT auth.uid())
        AND role = 'admin'
    );
$$;

-- ==========================================
-- AUTOMATIC POINTS FROM TRANSACTIONS
-- Rp10.000 = 1 POINT
-- ==========================================

CREATE OR REPLACE FUNCTION public.award_transaction_points()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
    earned INTEGER;
BEGIN
    earned := FLOOR(NEW.total_amount / 10000)::INTEGER;

    NEW.points_earned := earned;

    IF earned > 0 THEN
        UPDATE public.members
        SET points_balance = points_balance + earned
        WHERE id = NEW.member_id;

        INSERT INTO public.point_transactions (
            member_id, type, points_delta,
            reference_type, reference_id, description
        )
        VALUES (
            NEW.member_id, 'earn', earned,
            'transaction', NEW.id, 'Poin dari transaksi belanja'
        );
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER transaction_points_trigger
BEFORE INSERT ON public.transactions
FOR EACH ROW
EXECUTE FUNCTION public.award_transaction_points();

-- ==========================================
-- REDEEM POINTS
-- 100 POINTS = Rp10.000 DISCOUNT
-- ==========================================

CREATE OR REPLACE FUNCTION public.redeem_points(
    p_member_id UUID,
    p_points INTEGER
)
RETURNS NUMERIC
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
    current_points INTEGER;
BEGIN
    IF NOT public.is_admin() THEN
        RAISE EXCEPTION 'Unauthorized';
    END IF;

    IF p_points <= 0 OR p_points % 100 <> 0 THEN
        RAISE EXCEPTION 'Penukaran harus kelipatan 100 poin';
    END IF;

    SELECT points_balance INTO current_points
    FROM public.members
    WHERE id = p_member_id
    FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Member tidak ditemukan';
    END IF;

    IF current_points < p_points THEN
        RAISE EXCEPTION 'Saldo poin tidak mencukupi';
    END IF;

    UPDATE public.members
    SET points_balance = points_balance - p_points
    WHERE id = p_member_id;

    INSERT INTO public.point_transactions (
        member_id, type, points_delta, description
    )
    VALUES (
        p_member_id, 'redeem', -p_points,
        'Penukaran poin menjadi diskon'
    );

    RETURN (p_points / 100) * 10000;
END;
$$;

-- ==========================================
-- ORDER POINT AUTOMATION
-- ==========================================

CREATE OR REPLACE FUNCTION public.process_order_points()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
    matched_member UUID;
    earned INTEGER;
BEGIN
    -- Pesanan selesai: cari member berdasarkan nomor HP
    IF NEW.status = 'Selesai'
       AND OLD.status IS DISTINCT FROM 'Selesai' THEN

        SELECT id INTO matched_member
        FROM public.members
        WHERE phone = NEW.customer_phone;

        IF matched_member IS NOT NULL THEN
            -- 1 poin per pesanan yang selesai
            earned := 1;

            IF earned > 0 THEN
                UPDATE public.members
                SET points_balance = points_balance + earned
                WHERE id = matched_member;

                INSERT INTO public.point_transactions (
                    member_id, type, points_delta,
                    reference_type, reference_id, description
                )
                VALUES (
                    matched_member, 'order_earn', earned,
                    'order', NEW.id, 'Poin dari pesanan online'
                );
            END IF;

            NEW.member_id := matched_member;
            NEW.points_awarded := earned;
        END IF;
    END IF;

    -- Pesanan dibatalkan setelah poin diberikan
    IF NEW.status = 'Dibatalkan'
       AND OLD.status = 'Selesai'
       AND OLD.points_awarded > 0 THEN

        UPDATE public.members
        SET points_balance = GREATEST(
            0, points_balance - OLD.points_awarded
        )
        WHERE id = OLD.member_id;

        INSERT INTO public.point_transactions (
            member_id, type, points_delta,
            reference_type, reference_id, description
        )
        VALUES (
            OLD.member_id, 'order_reversal',
            -OLD.points_awarded, 'order', OLD.id,
            'Pembatalan poin pesanan'
        );

        NEW.points_awarded := 0;
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER order_points_trigger
BEFORE UPDATE OF status ON public.orders
FOR EACH ROW
EXECUTE FUNCTION public.process_order_points();

-- PROFILES
CREATE POLICY "Admin read profiles"
ON public.profiles FOR SELECT
TO authenticated
USING (public.is_admin() OR id = (SELECT auth.uid()));

CREATE POLICY "Admin manage profiles"
ON public.profiles FOR ALL
TO authenticated
USING (public.is_admin())
WITH CHECK (public.is_admin());

-- MEMBERS
CREATE POLICY "Admin manage members"
ON public.members FOR ALL
TO authenticated
USING (public.is_admin())
WITH CHECK (public.is_admin());

-- TRANSACTIONS
CREATE POLICY "Admin manage transactions"
ON public.transactions FOR ALL
TO authenticated
USING (public.is_admin())
WITH CHECK (public.is_admin());

-- POINT TRANSACTIONS
CREATE POLICY "Admin manage point transactions"
ON public.point_transactions FOR ALL
TO authenticated
USING (public.is_admin())
WITH CHECK (public.is_admin());

-- PRODUCTS
CREATE POLICY "Public view active products"
ON public.products FOR SELECT
TO anon, authenticated
USING (is_active = TRUE);

CREATE POLICY "Admin manage products"
ON public.products FOR ALL
TO authenticated
USING (public.is_admin())
WITH CHECK (public.is_admin());

-- ORDERS
CREATE POLICY "Public create orders"
ON public.orders FOR INSERT
TO anon, authenticated
WITH CHECK (status = 'Baru' AND points_awarded = 0);

CREATE POLICY "Admin manage orders"
ON public.orders FOR ALL
TO authenticated
USING (public.is_admin())
WITH CHECK (public.is_admin());

-- ORDER ITEMS
CREATE POLICY "Public create order items"
ON public.order_items FOR INSERT
TO anon, authenticated
WITH CHECK (
    EXISTS (
        SELECT 1
        FROM public.orders
        WHERE id = order_id AND status = 'Baru'
    )
);

CREATE POLICY "Admin manage order items"
ON public.order_items FOR ALL
TO authenticated
USING (public.is_admin())
WITH CHECK (public.is_admin());
